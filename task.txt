def run_su(cmd, timeout=15):
    try:
        r = subprocess.run(["su", "-c", cmd], capture_output=True, text=True,
                           timeout=timeout, errors="replace")
        return r.returncode == 0, r.stdout.strip(), r.stderr.strip()
    except subprocess.TimeoutExpired:
        return False, "", "timeout"
    except Exception as e:
        return False, "", str(e)

def _looks_like_url(t):
    t = t.strip()
    if not t or len(t) < 8: return False
    if t.startswith("http") or "://" in t: return True
    if re.match(r'^(www\.)?[a-z0-9-]+\.[a-z]{2,}([/?].*)?$', t, re.I): return True
    if re.search(r'(lootlabs|linkvertise|work\.ink|izen|zen\.lol|\.gg/|\.ink/|relay|platorelay)', t, re.I): return True
    return False

def tg_send(text, silent=False, retries=3):
    if not TG_BOT_TOKEN or not TG_CHAT_ID:
        return False
    url = f"https://api.telegram.org/bot{TG_BOT_TOKEN}/sendMessage"
    for i in range(retries):
        try:
            r = _HTTP.post(url, json={"chat_id": TG_CHAT_ID, "text": str(text)[:4000],
                                      "disable_notification": silent}, timeout=10)
            if r.status_code == 200: return True
            if r.status_code == 429:
                try:
                    ra = r.json().get("parameters", {}).get("retry_after", 5)
                except Exception:
                    ra = 5
                time.sleep(ra + 1); continue
            return False
        except Exception:
            time.sleep(2 + i * 2)
    return False

def utc_now():
    return datetime.now(timezone.utc) + timedelta(seconds=CLOCK_OFFSET[0])

def msk_now():
    return utc_now() + timedelta(hours=MSK_UTC_OFFSET)

def measure_clock_offset(apply=True):
    try:
        r = _HTTP.head("https://api.github.com", timeout=10)
        d = r.headers.get("Date")
        if d:
            server = eut.parsedate_to_datetime(d)
            if server.tzinfo is None:
                server = server.replace(tzinfo=timezone.utc)
            off = (server - datetime.now(timezone.utc)).total_seconds()
            if abs(off) > 120 and time.time() - _last_drift_alert[0] > 3600:
                log(f"🕐 Часы уходят на {off:+.0f}с — скорректировано")
                _last_drift_alert[0] = time.time()
            if apply: CLOCK_OFFSET[0] = off
            return off
    except Exception:
        pass
    return 0.0

def _hour_now():
    now = utc_now()
    return now.hour + now.minute / 60.0 + now.second / 3600.0

def utc_slot():
    h = _hour_now()
    if h < HARVEST_UTC_HOUR:
        return "farm", -1
    for b in range(HARVEST_BATCHES):
        if h < HARVEST_UTC_HOUR + (b + 1) * HARVEST_SLOT_MIN / 60.0:
            return "harvest", b
    return "farm", -1

def harvest_minutes_away():
    h = _hour_now()
    if h < HARVEST_UTC_HOUR:
        return (HARVEST_UTC_HOUR - h) * 60.0
    return (24.0 - h + HARVEST_UTC_HOUR) * 60.0

def minutes_to_next_boundary():
    h = _hour_now()
    bs = [HARVEST_UTC_HOUR] \
       + [HARVEST_UTC_HOUR + (b + 1) * HARVEST_SLOT_MIN / 60.0 for b in range(HARVEST_BATCHES)] \
       + [24.0]
    for b in bs:
        if h < b - 1e-9:
            return (b - h) * 60.0
    return 0.0

def _after_slots_now():
    return _hour_now() >= HARVEST_UTC_HOUR + HARVEST_BATCHES * HARVEST_SLOT_MIN / 60.0

def wait_out_slot(slot_tuple):
    while utc_slot() == slot_tuple:
        check_stop()
        sleep_stop(max(30.0, minutes_to_next_boundary() * 60 - 90))

def _proc_cmdline_contains(pid, fragment):
    try:
        pid = int(str(pid).strip())
        if pid <= 0:
            return False
        with open(f"/proc/{pid}/cmdline", "rb") as f:
            cmd = f.read().replace(b"\0", b" ").decode(errors="replace")
        return fragment in cmd
    except Exception:
        return False

def acquire_lock():
    try:
        if os.path.exists(PID_FILE):
            with open(PID_FILE) as f:
                old = f.read().strip()
            if old and old != str(os.getpid()) and _proc_cmdline_contains(old, "farm.py"):
                log(f"⛔ Уже запущен другой farm.py (PID {old}) — выхожу.")
                log("   Полная остановка: sh ~/farm_stop.sh | запуск: python farm.py")
                return False
        with open(PID_FILE, "w") as f:
            f.write(str(os.getpid()))
        return True
    except Exception as e:
        logv(f"⚠️ lock: {e}")
        return True

def release_lock():
    try:
        if os.path.exists(PID_FILE):
            with open(PID_FILE) as f:
                p = f.read().strip()
            if p == str(os.getpid()):
                os.remove(PID_FILE)
    except Exception:
        pass

def windows_track(pkg, code=None):
    with _WINDOWS_LOCK:
        _WINDOWS[pkg] = {"code": code, "ts": time.time(), "fails": 0}

def windows_untrack(pkg):
    with _WINDOWS_LOCK:
        _WINDOWS.pop(pkg, None)

def windows_clear():

        launch_private(pkg, code)

def windows_watchdog_loop():
    log(f"👁 Ватчдог окон: проверка каждые {WINDOW_CHECK_INTERVAL} сек (только клоны)")
    while True:
        try:
            time.sleep(WINDOW_CHECK_INTERVAL)
            if stop_requested():
                return
            with _WINDOWS_LOCK:
                targets = {k: dict(v) for k, v in _WINDOWS.items()}
            for pkg, info in targets.items():
                try:
                    if pkg_alive(pkg):
                        with _WINDOWS_LOCK:
                            if pkg in _WINDOWS:
                                _WINDOWS[pkg]["fails"] = 0
                        continue
                    if time.time() - info.get("ts", 0) < WINDOW_LAUNCH_GRACE:
                        continue
                    with _WINDOWS_LOCK:
                        if pkg not in _WINDOWS:
                            continue
                        fails = _WINDOWS[pkg].get("fails", 0) + 1
                        _WINDOWS[pkg]["fails"] = fails
                        _WINDOWS[pkg]["ts"] = time.time()
                        code = _WINDOWS[pkg].get("code")
                    if fails > WINDOW_MAX_RELANCHES:
                        log(f"👎 {pkg} вылетел {fails}× подряд — снимаю с ватчдога")
                        tg_send(f"🆘 {PHONE_ID}: {pkg} вылетает подряд ({fails}×) — "
                                f"окно больше не поднимаю", silent=True)
                        windows_untrack(pkg)
                        continue
                    _relaunch_window(pkg, code)
                except Exception as e:
                    logv(f"⚠️ watchdog[{pkg}]: {e}")
        except Exception as e:
            logv(f"⚠️ windows watchdog: {e}")

def windows_watchdog_start():
    if _WATCHDOG_STARTED[0]:
        return
    _WATCHDOG_STARTED[0] = True
    threading.Thread(target=windows_watchdog_loop, daemon=True).start()

def gh_api(method, path, body=None):
    url = f"https://api.github.com/repos/{GH_REPO}/contents/{path}"
    params = {"ref": GH_BRANCH} if method == "GET" else None
    headers = {"Authorization": f"Bearer {GH_TOKEN}",
               "Accept": "application/vnd.github+json", "User-Agent": PHONE_ID}
    try:
        r = _HTTP.request(method, url, params=params, json=body, headers=headers, timeout=25)
        try:
            j = r.json() if r.content else {}
        except Exception:
            j = {}
        if r.status_code in (401, 403):
            msg = (j.get("message") or "") if isinstance(j, dict) else ""
            log(f"⚠️ GitHub {r.status_code}: {msg[:120]}")
        return r.status_code, j
    except Exception as e:
        logv(f"⚠️ GitHub API: {e}")
        return 0, {}

def gh_get(path):
    st, j = gh_api("GET", path)
    if st == 200 and isinstance(j, dict):
        try:
            content = base64.b64decode(j.get("content", "")).decode()
        except Exception:
            content = ""
        return j.get("sha"), content
    return None, None

def gh_put(path, content, expect_create=False, silent=False, sha=None):
    body = {"message": f"autofarm [{PHONE_ID}] {path}",
            "content": base64.b64encode(content.encode()).decode(),
            "branch": GH_BRANCH}
    if sha:
        body["sha"] = sha
    st, j = gh_api("PUT", path, body)
    if st in (200, 201):
        if path.startswith(FLEET_DIR + "/") and isinstance(j, dict):
            _PUT_SHA_CACHE[path] = (j.get("content") or {}).get("sha")
        return True
    if expect_create and st == 422:
        return False
    if not silent:
        log(f"⚠️ PUT {path}: HTTP {st}")
    return False

def gh_delete(path):
    _PUT_SHA_CACHE.pop(path, None)
    st, j = gh_api("GET", path)
    if st == 404:
        return True
    if st == 0:
        return False
    sha = j.get("sha") if isinstance(j, dict) else None
    if not sha:
        return False
    st2, _j = gh_api("DELETE", path, {"message": f"autofarm [{PHONE_ID}] del {path}",
                                      "sha": sha, "branch": GH_BRANCH})
    return st2 in (200, 204)

def gh_list(path):
    dir_path = (path or "").strip("/")
    headers = {"Authorization": f"Bearer {GH_TOKEN}",
               "Accept": "application/vnd.github+json", "User-Agent": PHONE_ID}
    try:
        r = _HTTP.get(f"https://api.github.com/repos/{GH_REPO}/git/trees/{GH_BRANCH}",
                      headers=headers, timeout=25)
        if r.status_code != 200:
            return []
        sha_dir = next((t["sha"] for t in r.json().get("tree", [])
                        if t.get("path", "").strip("/") == dir_path and t.get("type") == "tree"), None)
        if not sha_dir:
            return []
        r2 = _HTTP.get(f"https://api.github.com/repos/{GH_REPO}/git/trees/{sha_dir}",
                       headers=headers, timeout=25)
        if r2.status_code != 200:
            return []
        j2 = r2.json()
        if j2.get("truncated"):
            log(f"⚠️ gh_list({dir_path}): дерево ОБРЕЗАНО")
        return [(t.get("path", ""), "") for t in j2.get("tree", []) if t.get("type") == "blob"]
    except Exception:
        return []

def phone_stats():
    if time.time() - _STATS_CACHE["ts"] < 120:
        return _STATS_CACHE["v"]
    bat = temp = ram = "?"
    ok, out, _ = run_su("dumpsys battery | grep -E 'level:|temperature:'; "
                        "grep MemAvailable /proc/meminfo", timeout=8)
    if ok and out:
        m = re.search(r'level:\s*(\d+)', out)
        if m: bat = m.group(1) + "%"
        m = re.search(r'temperature:\s*(\d+)', out)
        if m: temp = f"{int(m.group(1)) / 10:.0f}C"
        m = re.search(r'MemAvailable:\s*(\d+)', out)
        if m: ram = f"{int(m.group(1)) // 1024}MB"
    _STATS_CACHE["ts"] = time.time()
    _STATS_CACHE["v"] = (bat, temp, ram)
    return bat, temp, ram

def _cookie_hash(c):
    return hashlib.sha1(c.encode()).hexdigest()[:12]

def pool_reap_claims():
    now = utc_ts()
    for name, _ in gh_list("claims"):
        if not name.endswith(".json"):
            continue
        h = name[:-5]
        _s, content = gh_get(f"claims/{h}.json")
        try:
            meta = json.loads(content) if content else {}
        except Exception:
            meta = {}
        meta = meta or {}
        if now - meta.get("ts", 0) < POOL_CLAIM_TTL:
            continue
        if gh_get(f"pool/{h}.txt")[0]:
            gh_delete(f"claims/{h}.json")
            continue
        lost = meta.get("cookie", "") or ""
        if lost and len(lost) > 40:
            st, _u, _n = fetch_user_info_full(lost)
            if st == "ok":
                if gh_put(f"pool/{h}.txt", lost, expect_create=True, silent=True):
                    log(f"♻️ Кука {h} восстановлена из протухшего claim")
                    gh_delete(f"claims/{h}.json")
            elif st == "dead":
                logv(f"☠️ Кука {h} из claim мертва — списываю")
                gh_delete(f"claims/{h}.json")
            else:
                logv(f"⏳ reap {h}: API недоступен — claim оставлен")
        else:
            gh_delete(f"claims/{h}.json")

def pool_upload(cookie):
    if not (POOL_ENABLED and GH_TOKEN and GH_REPO) or not cookie or len(cookie) < 40:
        return False
    h = _cookie_hash(cookie)
    sha, _c = gh_get(f"pool/{h}.txt")
    if sha:
        logv(f"☁️ {h} уже в пуле (dup)")
        return "dup"
    ok = gh_put(f"pool/{h}.txt", cookie, expect_create=True, silent=True)
    if not ok:
        if gh_get(f"pool/{h}.txt")[0]:
            return "dup"
        log(f"⚠️ Не удалось загрузить {h} в пул")
        return False
    logv(f"☁️ Кука {h} → в пул")
    return True

def pool_count():
    if not (GH_TOKEN and GH_REPO):
        return -1
    return len([x for x, _ in gh_list("pool") if x.endswith(".txt")])

def pool_claim():
    if not (GH_TOKEN and GH_REPO):
        return None
    try:
        pool_reap_claims()
    except Exception as e:
        logv(f"⚠️ reap: {e}")
    claimed = {name[:-5] for name, _ in gh_list("claims") if name.endswith(".json")}
    now = utc_ts()
    for name, _sha in gh_list("pool"):
        h = name[:-4] if name.endswith(".txt") else name
        claim_path = f"claims/{h}.json"
        if h in claimed:
            _s, content = gh_get(claim_path)
            try:
                meta = json.loads(content) if content else {}
                ts = meta.get("ts", 0)
            except Exception:
                meta, ts = {}, 0
            if now - ts < POOL_CLAIM_TTL:
                continue
            gh_delete(claim_path)
            claimed.discard(h)
        if not gh_put(claim_path, json.dumps({"phone": PHONE_ID, "ts": now}), expect_create=True):
            continue
        _s, cookie = gh_get(f"pool/{h}.txt")
        if not (cookie and len(cookie.strip()) > 40):
            gh_delete(claim_path)
            continue
        cookie = cookie.strip()
        backed = False
        _csha, _c2 = gh_get(claim_path)
        if _csha:
            backed = gh_put(claim_path, json.dumps({"phone": PHONE_ID, "ts": now,
                                                    "cookie": cookie}), silent=True, sha=_csha)
        if backed:
            gh_delete(f"pool/{h}.txt")
            log(f"🎯 Кука захвачена из пула ({h})")
            return cookie
        gh_delete(claim_path)
        log(f"⚠️ Бэкап {h} в claim не удался — отпускаю")
    return None

def pool_consume_hash(h):
    gh_delete(f"claims/{h}.json")

def pool_consume(cookie):
    pool_consume_hash(_cookie_hash(cookie))

def pool_topup(need):
    got = []
    if need <= 0 or not (GH_TOKEN and GH_REPO):
        return got
    log(f"☁️ Пул: добираю {need} кук...")
    while need > 0:
        c = pool_claim()
        if not c:
            log("☁️ Пул пуст/недоступен")
            break
        st, uid, uname = fetch_user_info_full(c)
        if st == "ok":
            got.append(c); need -= 1
            log(f"✅ Из пула: {uname} | осталось: {need}")
        elif st == "dead":
            logv("☠️ Кука из пула мёртвая — списываю")
            pool_consume(c)
        else:
            log("⚠️ Проверка не удалась (сеть?) — возврат в пул")
            if pool_upload(c) in (True, "dup"):
                pool_consume(c)
            else:
                log("⚠️ Вернуть не удалось — claim оставлен, TTL вернёт кукy в пул")
            break
    return got

def get_field_length():
    xml = ui_dump()
    if not xml:
        return -1
    first = -1
    for m in re.finditer(r'<node[^>]*/?>', xml):
        attrs = dict(re.findall(r'([\w-]+)="([^"]*)"', m.group(0)))
        if "EditText" not in attrs.get("class", ""):
            continue
        ln = len(html_mod.unescape(attrs.get("text", "")))
        if attrs.get("focused") == "true" or attrs.get("password") == "true":
            return ln
        if first < 0:
            first = ln
    return first

def password_field_visible():
    xml = ui_dump()
    if not xml:
        return False
    return any("EditText" in dict(re.findall(r'([\w-]+)="([^"]*)"', m.group(0))).get("class", "")
               for m in re.finditer(r'<node[^>]*/?>', xml))

def wait_password_field(timeout=P_FIELD_WAIT, reserve_tap=TAP_X):
    deadline = time.time() + timeout
    poked = False
    while time.time() < deadline:
        if password_field_visible():
            logv("✅ Поле пароля открыто")
            return True
        if not poked and (deadline - time.time()) <= 6:
            tap(*reserve_tap)
            poked = True
        time.sleep(1)
    return False

def get_table_cols():
    try:
        con = sqlite3.connect(WORK_DB)

def _read_roblosecurity(db):
    if not run_su(f"test -f '{db}'")[0]:
        return None
    copy_db_to_work(db)
    con = None
    try:
        con = sqlite3.connect(WORK_DB)
        has = con.execute("SELECT name FROM sqlite_master WHERE type='table' "
                          "AND name='cookies'").fetchone()
        if not has:
            return None
        rows = con.execute("SELECT value FROM cookies WHERE name='.ROBLOSECURITY' "
                           "AND host_key LIKE '%roblox.com' ORDER BY expires_utc DESC").fetchall()
        for (val,) in rows:
            if val and len(val) > 40:
                return val.strip()
    except Exception:
        pass
    finally:
        if con:
            try: con.close()
            except Exception: pass
        clean_work_db()
    return None

def fetch_user_info_full(cookie):
    try:
        r = _HTTP.get("https://users.roblox.com/v1/users/authenticated",
                      headers={"Cookie": f".ROBLOSECURITY={cookie}", "User-Agent": "Roblox/Win32"},
                      timeout=15)

        if r.status_code == 401:
            return "dead", None, None
        return "error", None, None
    except Exception:
        return "error", None, None

def clean_dead_cookies():

    if not os.path.exists(COOKIES_F):
        return 0, 0
    raw = load_cookies_file()
    unique = list(dict.fromkeys(raw))
    dupes = len(raw) - len(unique)
    live, dead = [], 0
    for c in unique:
        st, uid, uname = fetch_user_info_full(c)
        if st == "ok":
            live.append(c)
        elif st == "dead":
            dead += 1
        else:
            live.append(c)

        with open(COOKIES_F + ".tmp", "w") as f:
            for c in live:
                f.write(c + "\n")
        os.replace(COOKIES_F + ".tmp", COOKIES_F)
    log(f"🧹 ЧИСТКА: живых {len(live)}, мёртвых {dead}, дублей {dupes}")
    return len(live), dead + dupes

def webview_version():
    for wv in ("com.google.android.webview", "com.android.webview",
               "com.google.android.trichromelibrary"):
        ok, out, _ = run_su(f"dumpsys package {wv} | grep -m1 versionName")
        if ok and out:
            m = re.search(r'versionName=(\d+)', out)
            if m:
                return int(m.group(1))
    return None

def _win_to_chrome_time(days=365):
    return int((time.time() + days * 86400) * 1_000_000) + 11644473600 * 1_000_000

            if wv and wv < 115:
                need_recreate = not (has_old and not has_new)
            else:
                need_recreate = not has_new
    if need_recreate:
        if not _native_create_one(pkg, db):

            finally:
                if con:
                    try: con.close()
                    except Exception: pass
    copy_db_to_work(db)
    con = None
    try:

        cur = con.cursor()
        cols_info = cur.execute("PRAGMA table_info(cookies)").fetchall()
        if not cols_info:
            return False
        now_us = _win_to_chrome_time(0)
        defaults = {
            "creation_utc": now_us, "top_frame_site_key": "", "host_key": ".roblox.com",
            "name": ".ROBLOSECURITY", "value": cookie, "encrypted_value": b"",
            "path": "/", "expires_utc": now_us + 946080000000000, "is_secure": 1,

            "is_persistent": 1, "priority": 1, "samesite": -1, "source_scheme": 2,

        clean_work_db()
        return True
    except Exception as e:
        log(f"❌ Инъекция {db}: {e}"); clean_work_db(); return False
    finally:

            try: con.close()
            except Exception: pass

def inject_prefs_identity(pkg, username, userid):
    prefs = f"/data/data/{pkg}/shared_prefs/prefs.xml"
    ok, content, _ = run_su(f"cat '{prefs}'")
    if not ok or not content:
        content = "<?xml version='1.0' encoding='utf-8' standalone='yes' ?>\n<map></map>"
    new = content
    for tag in ("username", "displayName"):
        if re.search(rf'<string\s+name="{tag}"\s*>', new):
            new = re.sub(rf'(<string\s+name="{tag}"\s*>)[^<]*(</string>)',
                         lambda m: m.group(1) + username + m.group(2), new)
        else:
            new = new.replace("</map>", f'<string name="{tag}">{username}</string></map>')
    if re.search(r'<long\s+name="userid_long"\s+value="', new):
        new = re.sub(r'(<long\s+name="userid_long"\s+value=")[^"]*("\s*/>)',
                     lambda m: m.group(1) + str(userid) + m.group(2), new)
    else:
        new = new.replace("</map>", f'<long name="userid_long" value="{userid}" /></map>')
    if new != content:
        escaped = new.replace("'", "'\\''")
        run_su(f"printf '%s' '{escaped}' > '{prefs}'")
        fix_ownership(pkg, f"/data/data/{pkg}/shared_prefs")

def gen_nick():
    letters = "".join(random.choice(string.ascii_lowercase) for _ in range(random.randint(8, 11)))
    return letters + str(random.randint(100, 9999))

def gen_pass(_nick):
    while True:
        p = "".join(random.choice(string.ascii_lowercase + string.ascii_uppercase + string.digits)
                    for _ in range(14))
        if any(c.isupper() for c in p) and any(c.islower() for c in p) and any(c.isdigit() for c in p):
            return p

RATE_LIMIT_MARKERS = (
    "too many attempts", "too many tries", "слишком много попыток",
    "слишком много", "try again later", "попробуйте позже",
    "too many requests", "rate limit", "превышено",
)

def check_rate_limit():
    xml = ui_dump()
    if not xml:
        return False
    texts = []
    for m in re.finditer(r'<node[^>]*/?>', xml):
        attrs = dict(re.findall(r'([\w-]+)="([^"]*)"', m.group(0)))
        t = html_mod.unescape(attrs.get("text", ""))
        if t: texts.append(t)
    txt = " | ".join(texts).lower()
    return any(marker in txt for marker in RATE_LIMIT_MARKERS)

                result, rl = None, False
                consec_errors += 1
                raised_exc = True
            else:
                consec_errors = 0
            finally:
                kill_main_only()
                if not _hard_reset_main(f"после попытки {done}"):
                    log("⚠️ Ресет после попытки не подтвердился")
            if raised_exc and consec_errors >= 3:
                raise RuntimeError("3 подряд упавших попытки — системная проблема")
            if rl:
                rate_limited = True
                break
            if result:
                if len(kept) < target_total:
                    kept.append(result)
                    save_cookie(result["cookie"], result["nick"], result["pass"])
                else:
                    logv(f"☁️ Себе хватает → {result['nick']} В ПУЛ")
                    if pool_upload(result["cookie"]) in (True, "dup"):
                        pooled += 1
                    else:
                        excess_pending.append(result)
                        _persist_excess(result)
            else:
                logv(f"❌ Попытка {done}/{total_attempts} провалилась")
        if out_of_time or (not pool_on and len(kept) >= target_total):
            break
        if round_num < MAX_ROUNDS:
            pause_min = RATE_LIMIT_PAUSE // 60 if rate_limited else ROUND_PAUSE_MIN
            log(f"⏸️ Пауза {pause_min} мин...")
            sleep_stop(pause_min * 60)
    if excess_pending:
        logv(f"☁️ Дозагрузка {len(excess_pending)} кук в пул (ретрай)...")
        for r in excess_pending:
            if pool_upload(r["cookie"]) in (True, "dup"):
                pooled += 1
    log(f"🏁 Рега: попыток {done}/{total_attempts} | себе {len(kept)} | в пул {pooled}")
    ui_cache_clear()
    return kept

def do_injection(cookies, only_empty=False):
    """Куки вписываются ТОЛЬКО в клоны-фармеры. main не трогаем."""
    if not cookies:
        log("☠️ Входить нечем")
        return 0, []
    pkgs = _farm_pkgs()
    if only_empty:
        pkgs = [p for p in pkgs if not has_session(p)]
    log(f"💉 Вход по кукам в КЛОНЫ: пакетов {len(pkgs)} | кук {len(cookies)}")
    ok_count = 0
    consumed = []
    recycle = []
    pairs = list(zip(pkgs, cookies))
    leftover = cookies[len(pairs):]
    for pkg, c in pairs:
        res = inject_pkg(pkg, c)
        if res == "unverified":
            consumed.append(c)
            log(f"⚠️ Не подтверждено (кука израсходована): {pkg}")
        elif res:
            ok_count += 1
            consumed.append(c)
            log(f"🎉 ЗАШЛО: {pkg} → {res}")
        else:
            st2, _u, _n = fetch_user_info_full(c)
            if st2 == "dead":
                consumed.append(c)
                log(f"☠️ Не зашло (кука мертва): {pkg}")
            else:
                recycle.append(c)
                log(f"⚠️ Не зашло (кука не тронута): {pkg}")
    for c in leftover:
        st2, _u, _n = fetch_user_info_full(c)
        if st2 == "dead":
            consumed.append(c)
        else:
            recycle.append(c)
    if len(cookies) < len(pkgs):
        empty = pkgs[len(cookies):]
        log(f"⚠️ Кук не хватило — без акка остались: {', '.join(empty) if empty else '—'}")
    archive_cookie_list(consumed, "инъекция завершена")
    return ok_count, recycle

def _drop_cookies_from_file(cookies):
    if not cookies or not os.path.exists(COOKIES_F):
        return
    drop = {_cookie_hash(c) for c in cookies}
    try:
        remaining = [c for c in load_cookies_file() if _cookie_hash(c) not in drop]
        with open(COOKIES_F + ".tmp", "w") as f:
            for c in remaining:
                f.write(c + "\n")
        os.replace(COOKIES_F + ".tmp", COOKIES_F)
    except Exception as e:
        log(f"🧹 ⚠️ Не убрал куки из cookies.txt: {e}")

def _recycle_cookies(recycle, claimed):
    rec_h = {_cookie_hash(c) for c in recycle}
    claimed_h = {_cookie_hash(c) for c in claimed}
    ok_h = set()
    for c in recycle:
        h = _cookie_hash(c)
        if pool_upload(c) in (True, "dup"):
            ok_h.add(h)
            _drop_cookies_from_file([c])
        else:
            log(f"⚠️ Кукy {h} не удалось вернуть в пул")
    for h in claimed_h:
        if h in rec_h and h not in ok_h:
            logv(f"♻️ Claim {h} оставлен — TTL вернёт кукy в пул")
            continue
        pool_consume_hash(h)

def _drain_local_to_pool():
    raw = load_cookies_file()
    if not raw:
        return
    uniq = list(dict.fromkeys(raw))
    ok = 0
    for c in uniq:
        if pool_upload(c) in (True, "dup"):
            ok += 1
            _drop_cookies_from_file([c])
    log(f"☁️ Дренаж cookies.txt: {ok}/{len(uniq)} → в пул")

def _termux_clip():
    for exe in ("termux-clipboard-get", "/data/data/com.termux/files/usr/bin/termux-clipboard-get"):
        try:
            res = subprocess.run([exe], capture_output=True, text=True, timeout=3)
            if res.returncode == 0:
                return res.stdout.strip()
        except Exception:
            continue
    return ""

_CLIP_CMDS = (
    "service call clipboard 2 s16 com.termux i32 0",
    "service call clipboard 2 s16 com.termux s16 com.termux i32 0",
    "service call clipboard 2 i32 0",
)

def _service_clip_text():
    for cmd in _CLIP_CMDS:
        ok, out, _ = run_su(cmd, timeout=5)
        if not ok:
            continue
        tokens = re.findall(r'\b[0-9a-fA-F]{8}\b', out)
        if not tokens:
            continue
        data = b''.join(bytes.fromhex(t)[::-1] for t in tokens)
        return data.decode('utf-16-le', errors='ignore').replace('\x00', ' ')
    return ""

def _paste_probe():

def _key_from_texts(texts):
    for t in texts:
        m = KEY_RE.search(t)
        if m:
            return m.group(1)
    candidates = [t.strip() for t in texts if _plausible_key(t)]
    if candidates:
        candidates.sort(key=len, reverse=True)
        log(f"⚠️ Фолбэк-ключ (самый длинный текст): {candidates[0][:60]}")
        return candidates[0]
    return None

def get_clipboard():
    t = _termux_clip()
    if t and _looks_like_url(t):
        return _normalize_url(t)
    m = re.search(r'https?://\S+', _service_clip_text())
    if m:
        return m.group(0)
    for t in _paste_probe():
        if _looks_like_url(t):
            return _normalize_url(t)
    return ""

def _plausible_key(t):
    if not t:
        return False
    if KEY_RE.search(t):
        return True
    t = t.strip()
    if _looks_like_url(t):
        return False
    return 10 <= len(t) <= 120 and " " not in t and "\n" not in t

def get_copied_key():
    time.sleep(2)
    t = _termux_clip()
    if t:
        m = KEY_RE.search(t)
        if m:
            return m.group(1)
        if _plausible_key(t):
            return t.strip()
    m = KEY_RE.search(_service_clip_text())
    if m:
        return m.group(1)
    k = _key_from_texts(get_visible_text())
    if k:
        return k
    return _key_from_texts(_paste_probe())

def get_bypassed_link(original_link, depth=0):
    try:
        r = _HTTP.get(ZEN_API_URL, params={"url": original_link},
                      headers={"x-api-key": ZEN_API_KEY}, timeout=25)
        if r.status_code == 429:
            if depth >= 3:
                return None
            time.sleep(60)
            return get_bypassed_link(original_link, depth + 1)
        data = r.json()
        bypassed = data.get("result") or data.get("destination") or data.get("link")
        if bypassed and bypassed.startswith("http"): return bypassed
        match = re.search(r'(https?://[^\s"]+)', r.text)
        if match: return match.group(1)
    except Exception as e:
        log(f"❌ Zen API: {e}")
    return None

}

def inject_url_via_ui(url):
    if not url: return False
    host = re.sub(r'^https?://', '', url).split('/')[0].split(':')[0].lower()
    escaped = url.replace("'", "'\\''")
    ok, _, _err = run_su(f"am start -a android.intent.action.VIEW -d '{escaped}'", timeout=10)
    if ok:
        time.sleep(5)
        texts = [t.lower() for t in get_visible_text()]
        if any(host in t for t in texts):
            return True
        if host in get_visible_url().lower():
            return True
    escaped2 = url.replace("'", "'\\''").replace('"', '\\"')
    run_su(f"echo '{escaped2}' | termux-clipboard-set", timeout=5)
    run_su(f"service call clipboard 1 s16 '{escaped2}'", timeout=5)
    tap(*OMNIBOX_TAP, 0.3)
    tap(*OMNIBOX_TAP, 0.3)
    tap(*OMNIBOX_TAP, 0.5)
    time.sleep(1)
    select_all_delete()
    run_su("input keyevent 279", timeout=5)
    time.sleep(1.5)
    vis = get_visible_url()
    if host in vis.lower() and len(vis) >= len(url) * 0.8:
        run_su("input keyevent 66", timeout=5)
        return True
    select_all_delete()
    all_ok = True
    for ch in [url[i:i + 150] for i in range(0, len(url), 150)]:
        safe = ch.replace("'", "'\\''").replace(" ", "%s")
        ok2, _, _ = run_su(f"input text '{safe}'", timeout=10)
        if not ok2:
            all_ok = False
            break
        time.sleep(0.15)
    if all_ok:
        run_su("input keyevent 66", timeout=5)
        return True
    select_all_delete()
    for ch in url:
        low = ch.lower()
        if ch in _URI_KEYS:
            kind, arg = _URI_KEYS[ch]
            run_su(f"input {kind} {arg}", timeout=3)
        elif low in KEYCODE_LOWER:
            code = KEYCODE_LOWER[low]
            run_su(f"input keycombination 59 {code}" if ch.isupper() else f"input keyevent {code}", timeout=3)
        elif low in KEYCODE_DIGIT:
            run_su(f"input keyevent {KEYCODE_DIGIT[low]}", timeout=3)
        else:
            safe = ch.replace("'", "'\\''")
            run_su(f"input text '{safe}'", timeout=3)
        time.sleep(0.02)
    run_su("input keyevent 66", timeout=5)
    time.sleep(1)
    return True

def run_bypass_cycle(max_attempts=None):
    n = max_attempts if (max_attempts and max_attempts > 0) else BYPASS_RETRIES
    try:
        for attempt in range(1, n + 1):
            check_stop()
            logv(f"🔁 Байпас {attempt}/{n}: фаза 1")
            kill_all_roblox()
            wake_screen()
            launch_delta()
            time.sleep(UI_INITIAL_DELAY)

            tap(*LINK_TAP, UI_STEP_DELAYS["LINK_TAP"])
            set_landscape()
            time.sleep(UI_STEP_DELAYS["SET_LANDSCAPE"])
            tap(*CLOSE_TRANSL_1, UI_STEP_DELAYS["CLOSE_TRANSL_1"])
            tap(*TRIGGER_AD, UI_STEP_DELAYS["TRIGGER_AD"])
            tap(*CLOSE_AD, UI_STEP_DELAYS["CLOSE_AD"])
            tap(*CONTINUE_1, UI_STEP_DELAYS["CONTINUE_1"])
            tap(*GOOGLE_ROW, UI_STEP_DELAYS["GOOGLE_ROW"])
            tap(*COPY_LINK, UI_STEP_DELAYS["COPY_LINK"])
            time.sleep(COPY_SETTLE_DELAY)

            copied_link = get_clipboard()
            if not _looks_like_url(copied_link):
                logv("⚠️ Ссылка не получена")
                time.sleep(3); continue
            copied_link = _normalize_url(copied_link)

            bypassed_link = get_bypassed_link(copied_link)
            if not bypassed_link:
                logv("⚠️ Байпас ссылки не удался")
                time.sleep(5); continue

            if not inject_url_via_ui(bypassed_link):
                logv("⚠️ Ввод ссылки не удался")
                time.sleep(3); continue

            time.sleep(UI_STEP_DELAYS["OPEN_ZEN_LINK"])
            tap(*CONTINUE_KEY, UI_STEP_DELAYS["CONTINUE_KEY"])
            tap(*CLOSE_TRANSL_2, UI_STEP_DELAYS["CLOSE_TRANSL_2"])
            tap(*COPY_CODE, UI_STEP_DELAYS["COPY_CODE"])
            time.sleep(3)
            key = get_copied_key()

            if not key:
                kill_all_roblox()
                time.sleep(7)
                continue

            log(f"🎯 КЛЮЧ ПОЛУЧЕН: {key[:50]}...")
            state["last_key"] = key
            state["last_key_time"] = utc_ts()
            save_state()
            write_key_via_ui(key)
            kill_all_roblox()
            return True
        log(f"💥 Байпас не дал ключа за {n} попыток")
        return False
    finally:
        set_portrait()
        ui_cache_clear()

def do_bypass_once(max_attempts=None, force=False):
    """🔑 v6.5: force=True (режим farm / рестарт после вылета) —
    байпас ВСЕГДА, БЕЗ проверок свежести ключа.
    Обычный вызов — антидубль 25 мин."""
    set_phase("KEY")
    if force:
        log("🔑 ПРИНУДИТЕЛЬНЫЙ байпас — свежесть ключа НЕ проверяю")
    else:
        lk = state.get("last_key_time", 0) or 0
        age_min = (utc_ts() - lk) / 60.0 if lk else 1e9
        if age_min < BYPASS_MIN_GAP_MIN:
            logv(f"🔑 Ключ свежий ({age_min:.0f} мин) — байпас не нужен")
            return True
    n = max_attempts if (max_attempts and max_attempts > 0) else BYPASS_RETRIES
    log(f"🔑 Байпас: фаза 1 ×{n} → фаза 2 ×{KEY_INJECT_RETRIES}")
    write_delta_lua(NOOP_LUA)
    pre_ts = state.get("last_key_time", 0) or 0
    try:
        run_bypass_cycle(max_attempts=n)
        if (state.get("last_key_time", 0) or 0) > pre_ts and state.get("last_key"):
            inject_key_simple(state["last_key"])
            log("✅ Байпас завершён — новый ключ записан")
            return True
        log("❌ Байпас НЕ дал нового ключа — работаем на старом")
        return False
    finally:
        write_delta_lua(FARM_LUA_CONTENT)
        ui_cache_clear()

_SHARE_CODE_RE = re.compile(r'(?:privateServerLinkCode=|share\?code=)([A-Za-z0-9_\-]+)')

def _parse_servers_text(text):
    text = (text or "").lstrip("\ufeff").strip()
    if not text:
        return []
    try:
        data = json.loads(text)
    except (ValueError, TypeError):
        data = None
    if data is not None:
        if isinstance(data, dict) and isinstance(data.get("servers"), list):
            data = data["servers"]
        if isinstance(data, list):
            return data
        if isinstance(data, dict) and data.get("code"):
            return [data]
        log("⚠️ servers: JSON распознан, но структура неясна")
        return []
    out = []
    for raw in text.splitlines():
        line = raw.strip()
        if not line or line.startswith("#") or line.startswith("//"):
            continue
        owner = ""
        for sep in ("|", ";", "\t"):
            if sep in line:
                left, right = line.split(sep, 1)
                left, right = left.strip(), right.strip()
                if left and right:
                    line, owner = left, right
                    break
        m = _SHARE_CODE_RE.search(line)
        if m:
            line = m.group(1)
        else:
            parts = line.replace(",", " ").split()
            if parts:
                line = parts[0]
                if not owner and len(parts) > 1:
                    owner = parts[1]
        if line and len(line) >= 4:
            out.append({"code": line, "owner": owner} if owner else line)
    return out

def load_servers():
    if not SERVERS_JSON_URL:
        log("⚠️ SERVERS_JSON_URL не задан")
        return []
    text = ""
    m = re.match(r'https://raw\.githubusercontent\.com/([^/]+)/([^/]+)/(?:refs/heads/)?([^/]+)/(.+)',
                 SERVERS_JSON_URL)
    if m and GH_TOKEN:
        user, repo, branch, path = m.groups()
        try:
            r2 = _HTTP.get(f"https://api.github.com/repos/{user}/{repo}/contents/{path}",
                           params={"ref": branch},
                           headers={"Authorization": f"Bearer {GH_TOKEN}",
                                    "Accept": "application/vnd.github.raw",
                                    "User-Agent": PHONE_ID},
                           timeout=15)
            if r2.status_code == 200:
                text = r2.text
            else:
                logv(f"⚠️ servers API: HTTP {r2.status_code}")
        except Exception as e:
            logv(f"⚠️ servers API: {e}")
    if not text:
        try:
            r = _HTTP.get(f"{SERVERS_JSON_URL}?t={int(time.time())}", timeout=15)
            if r.status_code == 200:
                text = r.text
        except Exception as e:
            logv(f"⚠️ servers raw: {e}")
    if not text or not text.strip():
        log("⚠️ servers: файл пуст или недоступен")
        return []
    servers = _parse_servers_text(text)
    logv(f"📄 servers: {len(servers)} записей" if servers
         else "⚠️ servers: не распознано ни одной приватики")
    return servers

def monitor_lock_read():
    sha, content = gh_get(MONITOR_LOCK_PATH)
    if not sha:
        return None, None
    try:
        return json.loads(content), sha
    except Exception:
        return {}, sha

def _lock_write(meta, sha):
    return gh_put(MONITOR_LOCK_PATH, json.dumps(meta), silent=True, sha=sha)

def monitor_try_lock():
    if not (GH_TOKEN and GH_REPO and TG_BOT_TOKEN):
        return False
    body = json.dumps({"phone": PHONE_ID, "ts": utc_ts()})
    if gh_put(MONITOR_LOCK_PATH, body, expect_create=True, silent=True):
        return True
    meta, sha = monitor_lock_read()
    if not meta or not meta.get("phone"):
        if sha:
            logv("🔧 Битый monitor_lock — пересоздаю")
            gh_delete(MONITOR_LOCK_PATH)
        return gh_put(MONITOR_LOCK_PATH, body, expect_create=True, silent=True)
    if meta.get("phone") == PHONE_ID:
        if utc_ts() - meta.get("ts", 0) < MONITOR_LOCK_TTL:
            monitor_touch_lock()
            return True
        gh_delete(MONITOR_LOCK_PATH)
        return gh_put(MONITOR_LOCK_PATH, body, expect_create=True, silent=True)
    if utc_ts() - meta.get("ts", 0) < MONITOR_LOCK_TTL:
        return "BUSY"
    gh_delete(MONITOR_LOCK_PATH)
    return gh_put(MONITOR_LOCK_PATH, body, expect_create=True, silent=True)

def monitor_touch_lock():
    meta, sha = monitor_lock_read()
    if not sha:
        return monitor_try_lock()
    if not meta or meta.get("phone") != PHONE_ID:
        return False
    meta["ts"] = utc_ts()
    if _TG_OFFSET[0]:
        meta["tg_offset"] = _TG_OFFSET[0]
    if _lock_write(meta, sha):
        return True
    meta2, sha2 = monitor_lock_read()
    if sha2 and meta2 and meta2.get("phone") == PHONE_ID:
        meta2["ts"] = utc_ts()
        if _TG_OFFSET[0]:
            meta2["tg_offset"] = _TG_OFFSET[0]
        return _lock_write(meta2, sha2)
    return False

def _monitor_save_offset():
    meta, sha = monitor_lock_read()
    if not sha or not meta or meta.get("phone") != PHONE_ID:
        return False
    meta["ts"] = utc_ts()
    meta["tg_offset"] = _TG_OFFSET[0]
    return _lock_write(meta, sha)

    return "\n".join(lines)

_COOKIE_IN_LINE_RE = re.compile(
    r"\.?ROBLOSECURITY[\"']?\s*[:=\t]\s*[\"']?([^\s\"',;]+)")

def _cookie_like(v):
    return len(v) >= TG_COOKIE_MIN_LEN and \
           ("|" in v or "_" in v or len(v) >= 100)

def extract_cookies_from_text(text):
    out = []
    for raw in (text or "").splitlines():
        line = raw.strip()
        if not line or line.startswith("/"):
            continue
        found = False
        for m in _COOKIE_IN_LINE_RE.finditer(line):
            val = m.group(1).rstrip("\"',;").strip("|")
            if len(val) >= 40:
                out.append(val)
                found = True
        if found:
            continue
        line = line.strip("`\"',; |\t")
        if not line or line[0] in "[{\"":
            continue
        if "=" in line and "://" not in line:
            k, v = line.split("=", 1)
            v = v.strip()
            if k and _cookie_like(v) and " " not in v and "\t" not in v:
                out.append(v)
                continue
        if (" " in line) or ("\t" in line) or ("://" in line) or line.startswith("💚"):
            continue
        if len(line) >= TG_COOKIE_MIN_LEN:
            out.append(line)
    return list(dict.fromkeys(out))

def _tg_download_file_cookies(doc):
    try:
        r = _HTTP.get(f"https://api.telegram.org/bot{TG_BOT_TOKEN}/getFile",
                      params={"file_id": doc["file_id"]}, timeout=15)
        if r.status_code != 200:
            log(f"📎 getFile: HTTP {r.status_code}")
            return []
        fp = r.json().get("result", {}).get("file_path")
        if not fp:
            return []
        r2 = _HTTP.get(f"https://api.telegram.org/file/bot{TG_BOT_TOKEN}/{fp}", timeout=60)
        if r2.status_code != 200:
            log(f"📎 download: HTTP {r2.status_code}")
            return []
        text = r2.text
    except Exception as e:
        log(f"📎 download: {e}")
        return []
    out = []
    for line in text.splitlines():
        line = line.strip()
        if not line or line.startswith(("#", "//")):
            continue
        out.extend(extract_cookies_from_text(line))
    s = text.strip()
    if len(out) <= 1 and (s.startswith("[") or s.startswith("{")):
        try:
            data = json.loads(s)
        except Exception:
            data = None
        if isinstance(data, list):
            for x in data:
                if isinstance(x, str) and len(x.strip()) >= 40:
                    out.append(x.strip())
                elif isinstance(x, dict):
                    for v in x.values():
                        if isinstance(v, str) and len(v.strip()) >= 40:
                            out.append(v.strip())
        elif isinstance(data, dict):
            for v in data.values():
                if isinstance(v, str) and len(v.strip()) >= 40:
                    out.append(v.strip())
    return list(dict.fromkeys(out))

def _pool_ingest_cookies(cookies, source_label):

            + (f", дублей {dup}" if dup else "")
            + (f", ошибок {err}" if err else "")
            + f" | в пуле: {n if n >= 0 else '?'}")

def _handle_cookie_file(msg, doc, drain=False):
    fname = (doc.get("file_name") or "").lower()
    mime = (doc.get("mime_type") or "").lower()
    if not (fname.endswith((".txt", ".csv", ".json", ".log"))
            or "text" in mime or "json" in mime):
        logv(f"📎 Файл «{doc.get('file_name')}» не текстовый — пропущен")
        return
    d = msg.get("date", 0) or 0
    if drain and d and (utc_ts() - d) > 24 * 3600:
        return
    cookies = _tg_download_file_cookies(doc)
    if not cookies:
        cap = (msg.get("caption") or "").strip()
        if cap:
            cookies = extract_cookies_from_text(cap)
    if not cookies:
        logv("📎 Кук в файле не нашёл")
        return
    log(_pool_ingest_cookies(cookies, f"файл «{doc.get('file_name')}»"))

def _monitor_poll_loop():
    meta, _sha = monitor_lock_read()
    offset = (meta or {}).get("tg_offset", 0) or 0
    if offset:
        ups, ok = _tg_get(offset)
        if not ok and not _TG_CONFLICT[0]:
            log("⚠️ Сохранённый tg_offset невалиден — дренаж с нуля")
            offset = 0
    if not offset:
        log("🧹 Дренаж старых апдейтов")
        fails = 0
        while True:
            ups, ok = _tg_get(offset)
            if not ok:
                fails += 1
                if fails >= 10 or _TG_CONFLICT[0]:
                    break
                time.sleep(5)
                continue
            fails = 0
            if not ups:
                break
            for u in ups:
                _process_message(u.get("message") or {}, drain=True)
            offset = max([offset] + [u.get("update_id", 0) for u in ups]) + 1
            if len(ups) < 100:
                break
        logv(f"🧹 Дренаж завершён (offset={offset})")
    _TG_OFFSET[0] = offset
    last_touch = time.time()
    last_off_save = time.time()
    while True:
        if _TG_CONFLICT[0]:
            _TG_CONFLICT[0] = False
            log("🔒 TG 409 — сохраняю offset и отпускаю лок")
            _monitor_save_offset()
            gh_delete(MONITOR_LOCK_PATH)
            return
        if time.time() - last_touch >= MONITOR_LOCK_TOUCH:
            if not monitor_touch_lock():
                log("🔒 Лок потерян — ухожу")
                _monitor_save_offset()
                return
            last_touch = time.time()
        ups, ok = _tg_get(offset)
        if not ok:
            time.sleep(5)
            continue
        for u in ups:
            offset = max(offset, u.get("update_id", 0)) + 1
            _TG_OFFSET[0] = offset
            if os.environ.get("DEBUG_TG"):
                m0 = u.get("message") or {}
                log(f"📥 chat={(m0.get('chat') or {}).get('id')} doc={bool(m0.get('document'))} "
                    f"text={(m0.get('text') or m0.get('caption') or '')[:40]!r}")
            _process_message(u.get("message") or {})
        if ups and time.time() - last_off_save >= 30:
            if _monitor_save_offset():
                last_off_save = time.time()
        time.sleep(1)

def _is_browser_crash(e):
    if TargetClosedError is not None and isinstance(e, TargetClosedError):
        return True
    s = f"{type(e).__name__}: {e}"
    return ("Target page, context or browser has been closed" in s
            or "TargetClosedError" in s
            or "browser has been closed" in s.lower()
            or "browser has crashed" in s.lower()
            or "Target crashed" in s)

# ---------- системный timezone ----------

def get_os_timezone():
    if sys.platform == "win32":
        try:
            r = subprocess.run(["tzutil", "/g"], capture_output=True,
                               text=True, timeout=10)
            return r.stdout.strip() or None
        except Exception:
            return None
    try:
        return time.tzname[0]
    except Exception:
        return None

def set_os_timezone(tz):
    """IANA ('Europe/Berlin') ИЛИ Windows-ID — Windows-ID напрямую."""
    if not tz:
        return False
    if sys.platform == "win32":
        win_id = tz if "/" not in tz else IANA_TO_WINDOWS_TZ.get(tz)
        if not win_id:
            return False
        try:
            r = subprocess.run(["tzutil", "/s", win_id], capture_output=True,
                               text=True, timeout=10)
            if r.returncode != 0:
                log(f"⚠️  tzutil отказал ({(r.stderr or '').strip()[:80]}) — "
                    f"запусти терминал ОТ ИМЕНИ АДМИНИСТРАТОРА", "WARN")
                return False
            return True
        except Exception:
            return False
    try:
        import os as _os
        _os.environ["TZ"] = tz
        time.tzset()
        return True
    except Exception:
        return False

def restore_os_timezone(orig):
    if sys.platform == "win32" and orig:
        return set_os_timezone(orig)
    return False

def _tz_offset_min(tz_name):
    if not tz_name or ZoneInfo is None:
        return None
    try:
        return int(datetime.now(ZoneInfo(tz_name))
                   .utcoffset().total_seconds() // 60)
    except Exception:
        return None

def _intl_offset_min(page, tz_name):
    """UTC-смещение произвольной IANA-зоны силами браузера (Intl)."""
    if not tz_name:
        return None
    js = """
(tz) => {
    try {
        const parts = new Intl.DateTimeFormat('en-US',
            {timeZone: tz, timeZoneName: 'longOffset'}).formatToParts(new Date());
        const p = parts.find(x => x.type === 'timeZoneName');
        if (!p) return null;
        const m = p.value.match(/GMT([+-])(\\d{1,2})(?::(\\d{2}))?/);
        if (!m) return null;
        const sign = (m[1] === '-') ? -1 : 1;
        return sign * (parseInt(m[2], 10) * 60 + parseInt(m[3] || '0', 10));
    } catch (e) { return null; }
}
"""
    try:
        val = page.evaluate(js, tz_name)
        return int(val) if val is not None else None
    except Exception:
        return None

def make_screen(sw, sh):
    if Screen is None:
        return None
    candidates = (
        {"max_x": sw, "max_y": sh, "min_x": sw, "min_y": sh},
        {"max_x": sw, "max_y": sh},
        {"min_w": sw, "max_w": sw, "min_h": sh, "max_h": sh},
        {"min_width": sw, "max_width": sw,
         "min_height": sh, "max_height": sh},
        {"width": sw, "height": sh},
    )
    for kw in candidates:
        try:
            return Screen(**kw)
        except Exception:
            continue
    return None

def build_proxy(session):
    if PROXY_MODE == "static":
        user, pwd = STATIC_USER, STATIC_PASS
        if STATIC_USER_TEMPLATE and "{session}" in STATIC_USER_TEMPLATE:
            user = STATIC_USER_TEMPLATE.format(session=session)
        if STATIC_PASS_TEMPLATE and "{session}" in STATIC_PASS_TEMPLATE:
            pwd = STATIC_PASS_TEMPLATE.format(session=session)
        return {"server": STATIC_SERVER, "username": user, "password": pwd}
    password = PROXY_PASS_TEMPLATE.format(session=session,
                                          lifetime=PROXY_LIFETIME_MIN)
    return {"server": f"http://{PROXY_HOST}:{PROXY_PORT}",
            "username": PROXY_USER, "password": password}

def new_session_id():
    return "".join(secrets.choice(string.ascii_lowercase + string.digits)
                   for _ in range(PROXY_SESSION_LEN))

def _proxy_get(url, proxy, timeout=PROXY_PREFLIGHT_TIMEOUT):
    """GET через HTTP-прокси, preemptive auth + UA браузера."""
    server = proxy["server"]
    if server.startswith("socks"):
        raise RuntimeError("preflight не ходит через SOCKS-схему")
    creds = base64.b64encode(
        f"{proxy.get('username', '')}:{proxy.get('password', '')}"
        .encode()).decode()

    class _ProxyAuth(urllib.request.BaseHandler):
        def http_request(self, req):
            req.add_unredirected_header("Proxy-Authorization",
                                        f"Basic {creds}")
            return req
        https_request = http_request

    req = urllib.request.Request(url, headers={
        "User-Agent": UA_HEADER,
        "Accept": "application/json, */*",
    })
    opener = urllib.request.build_opener(
        urllib.request.ProxyHandler({"http": server, "https": server}),
        _ProxyAuth())
    return opener.open(req, timeout=timeout).read().decode("utf-8", "replace")

def _geo_lookup(ip, proxy):
    providers = (
        GEO_JSON_URL,
        "https://ipwho.is/" + (ip or ""),
        "https://api.ip.sb/geoip/" + (ip or ""),
    )
    for url in providers:
        try:
            g = json.loads(_proxy_get(url, proxy))
        except Exception as e:
            dbg(f"geo-провайдер {url} упал: {e.__class__.__name__}")
            continue
        if not isinstance(g, dict):
            continue
        cc = (g.get("country_code") or g.get("countryCode") or "").upper()
        tz = g.get("timezone")
        if isinstance(tz, dict):
            tz = tz.get("id")
        if not cc and not tz:
            continue
        return {"ip": g.get("ip") or ip,
                "country": g.get("country") or g.get("country_name"),
                "countryCode": cc,
                "city": g.get("city"),
                "timezone": tz}
    return None

def detect_per_request_rotation(proxy):
    """ДВА запроса ipify с паузой. IP разный → ротация per-request."""
    if not ROTATION_PROBE:
        return
    try:
        ip1 = _proxy_get("https://api.ipify.org", proxy).strip()
    except Exception:
        return
    time.sleep(ROTATION_PROBE_PAUSE_S)
    try:
        ip2 = _proxy_get("https://api.ipify.org", proxy).strip()
    except Exception:
        return
    if ip1 and ip2 and ip1 != ip2:
        raise RotationDetected(
            f"за {ROTATION_PROBE_PAUSE_S}с IP сменился {ip1} → {ip2}")

def preflight_proxy(proxy):
    """→ (exit_ip, geo) через сам прокси. Бросает RotationDetected."""
    if not PROXY_PREFLIGHT:
        return None, None
    try:
        ip = _proxy_get("https://api.ipify.org", proxy).strip()[:64]
    except Exception as e:
        log(f"⚠️  Preflight: прокси не отвечает ({e.__class__.__name__}: {e})",
            "WARN")
        log("   Проверь: креды (создай список прокси в ЛК!), "
            "PROXY_SCHEME, порт открыт?", "WARN")
        return None, None
    detect_per_request_rotation(proxy)
    geo = _geo_lookup(ip, proxy)
    if geo is None:
        log("⚠️  Preflight: все geo-провайдеры недоступны через прокси",
            "WARN")
    return ip, geo

def preflight_with_filter(prev_ip=None):
    """→ (proxy, session, exit_ip, geo)."""
    def _passes_country(geo):
        if not geo or not geo.get("countryCode"):
            return True
        cc = geo["countryCode"]
        if ALLOWED_COUNTRIES and cc not in ALLOWED_COUNTRIES:
            return False
        if cc in BLOCKED_COUNTRIES:
            return False
        return True

    if PROXY_MODE == "static":
        has_sessions = bool(
            (STATIC_USER_TEMPLATE and "{session}" in STATIC_USER_TEMPLATE)
            or (STATIC_PASS_TEMPLATE and "{session}" in STATIC_PASS_TEMPLATE))
        proxy, session, exit_ip, geo = None, "static", None, None
        for i in range(1, IP_ROLL_MAX_TRIES + 1):
            session = new_session_id() if has_sessions else "static"
            proxy = build_proxy(session)
            exit_ip, geo = preflight_proxy(proxy)
            if not exit_ip:
                return proxy, session, exit_ip, geo
            if not _passes_country(geo):
                log(f"↻ IP {exit_ip} ({geo.get('countryCode')}) не прошёл "
                    f"фильтр стран — перекат ({i}/{IP_ROLL_MAX_TRIES})")
                time.sleep(IP_ROLL_PAUSE_S)
                continue
            if NEW_IP_PER_ACCOUNT and prev_ip and exit_ip == prev_ip:
                if has_sessions:
                    log(f"↻ Сессия дала тот же IP ({exit_ip}) — "
                        f"перегенерирую ({i}/{IP_ROLL_MAX_TRIES})")
                else:
                    log(f"↻ IP не сменился ({exit_ip}) — жду ротацию пула "
                        f"({i}/{IP_ROLL_MAX_TRIES})")
                time.sleep(IP_ROLL_PAUSE_S)
                continue
            return proxy, session, exit_ip, geo
        if prev_ip and exit_ip == prev_ip:
            log(f"⚠️  IP удержался ({exit_ip}) за {IP_ROLL_MAX_TRIES} "
                f"попыток", "WARN")
        return proxy, session, exit_ip, geo

    proxy, session, exit_ip, geo = None, "-", None, None
    for i in range(1, PREFLIGHT_MAX_TRIES + 1):
        session = new_session_id()
        proxy = build_proxy(session)
        exit_ip, geo = preflight_proxy(proxy)
        if not geo or not geo.get("countryCode"):
            return proxy, session, exit_ip, geo
        cc = geo["countryCode"]
        if ALLOWED_COUNTRIES and cc not in ALLOWED_COUNTRIES:
            log(f"↻ IP {exit_ip} ({cc}) не в ALLOWED — перегенерирую "
                f"({i}/{PREFLIGHT_MAX_TRIES})")
            continue
        if cc in BLOCKED_COUNTRIES:
            log(f"↻ IP {exit_ip} ({cc}) в BLOCKED — перегенерирую "
                f"({i}/{PREFLIGHT_MAX_TRIES})")
            continue
        if NEW_IP_PER_ACCOUNT and prev_ip and exit_ip == prev_ip:
            log(f"↻ IP совпал с предыдущим аккаунтом — новая сессия "
                f"({i}/{PREFLIGHT_MAX_TRIES})")
            continue
        return proxy, session, exit_ip, geo
    log("⚠️  Фильтр/ротация: не подобрал IP за лимит — работаю с тем, "
        "что есть", "WARN")
    return proxy, session, exit_ip, geo

def log_exit_ip(page, expected_ip=None):
    """Проверка утечек. TZ-матч трёхуровневый:
      1) UTC-смещения равны; 2) один Windows-ID; 3) имена равны.
    → (tz_ok, browser_tz, proxy_tz)."""
    if not VERIFY_PROXY_IP:
        return (True, None, None)

    try:
        page.goto(CHECK_IP_URL, wait_until="domcontentloaded", timeout=15000)
        exit_ip = page.locator("body").inner_text().strip()[:64]
        log(f"🌐 Выходной IP: {exit_ip}")
        if expected_ip and exit_ip != expected_ip:
            log(f"⚠️  IP сменился между preflight ({expected_ip}) и браузером "
                f"({exit_ip}) — пул ротирует IP!", "WARN")
    except Exception as e:
        log(f"⚠️  Не проверил IP ({e.__class__.__name__})", "WARN")
        return (True, None, None)

    try:
        webrtc_ok = page.evaluate(
            "() => (typeof RTCPeerConnection === 'undefined' && "
            "       typeof webkitRTCPeerConnection === 'undefined')")
        if webrtc_ok:
            log("🔒 WebRTC: заблокирован (утечек нет)")
        else:
            log("⚠️  WebRTC: доступен в JS!", "WARN")
    except Exception:
        pass

    tz_ok, browser_tz, proxy_tz = True, None, None
    try:
        browser_tz = page.evaluate(
            "() => Intl.DateTimeFormat().resolvedOptions().timeZone")
        browser_off = -int(page.evaluate(
            "() => new Date().getTimezoneOffset()"))
        page.goto(GEO_JSON_URL, wait_until="domcontentloaded", timeout=15000)
        geo = json.loads(page.locator("body").inner_text())
        proxy_tz = geo.get("timezone")
        proxy_city = geo.get("city") or "—"
        proxy_country = geo.get("country") or "—"

        proxy_off = _tz_offset_min(proxy_tz)
        if proxy_off is None:
            proxy_off = _intl_offset_min(page, proxy_tz)

        win_b = IANA_TO_WINDOWS_TZ.get(browser_tz)
        win_p = IANA_TO_WINDOWS_TZ.get(proxy_tz)
        same_win = win_b is not None and win_b == win_p

        if proxy_off is not None and browser_off == proxy_off:
            log(f"🔒 Timezone совпадает (UTC{browser_off // 60:+d}): "
                f"{browser_tz} ≈ {proxy_tz} "
                f"({proxy_city}, {proxy_country})")
        elif same_win:
            log(f"🔒 Timezone совпадает (одна зона Windows "
                f"'{win_b}'): {browser_tz} ≈ {proxy_tz} "
                f"({proxy_city}, {proxy_country})")
        elif browser_tz == proxy_tz:
            log(f"🔒 Timezone совпадает (по имени): {browser_tz} "
                f"({proxy_city}, {proxy_country})")
        else:
            tz_ok = False
            off_str = (f"(UTC{browser_off // 60:+d}) " if proxy_off is not None
                       else "")
            log(f"🚨 TIMEZONE УТЕЧКА! Браузер: {browser_tz} {off_str}"
                f"vs прокси: {proxy_tz}", "ERROR")
    except Exception:
        pass

    try:
        browser_locale = page.evaluate(
            "() => navigator.language || navigator.languages[0]")
        log(f"🗣  Locale браузера: {browser_locale}")
    except Exception:
        pass

    return (tz_ok, browser_tz, proxy_tz)

def make_fingerprint():
    sw, sh = random.choice(SCREEN_POOL)
    scale = random.uniform(WINDOW_SCALE_MIN, WINDOW_SCALE_MAX)
    ww = max(800, int(sw * scale) // 2 * 2)
    wh = max(600, int(sh * scale) // 2 * 2)
    ww = min(ww, sw)
    wh = min(wh, sh)
    return {"screen": (sw, sh), "window": (ww, wh),
            "cores": random.choice(CPU_CORES_POOL)}

def random_password():
    while True:
        pwd = "".join(secrets.choice(
            string.ascii_letters + string.digits + "!@#$"
        ) for _ in range(16))
        if (any(c.isupper() for c in pwd)
                and any(c.islower() for c in pwd)
                and any(c.isdigit() for c in pwd)):
            return pwd

MOUSE = {"x": None, "y": None}

def _bezier(p0, p1, p2, t):
    x = (1 - t) ** 2 * p0[0] + 2 * (1 - t) * t * p1[0] + t ** 2 * p2[0]
    y = (1 - t) ** 2 * p0[1] + 2 * (1 - t) * t * p1[1] + t ** 2 * p2[1]
    return x, y

def _move_along(page, p0, p1, p2):
    dist = math.hypot(p2[0]-p0[0], p2[1]-p0[1])
    steps = random.randint(
        max(MOUSE_MIN_STEPS, int(dist / MOUSE_CURVE_MIN_DIV)),
        max(MOUSE_MIN_STEPS + 3, int(dist / MOUSE_CURVE_MAX_DIV)))
    for i in range(1, steps + 1):
        t = i / steps
        te = t * t * (3 - 2 * t)
        te += random.uniform(-0.03, 0.03) * (1 - abs(2*t - 1))
        te = max(0.0, min(1.0, te))
        bx, by = _bezier(p0, p1, p2, te)
        jf = MOUSE_JITTER * (1 + (MOUSE_JITTER_MID_BOOST - 1)
                             * math.sin(math.pi * t))
        bx += random.uniform(-jf, jf)
        by += random.uniform(-jf, jf)
        try:
            page.mouse.move(bx, by)
        except Exception:
            pass
        step_ms = random.randint(MOUSE_STEP_MIN_MS, MOUSE_STEP_MAX_MS)
        if t > 0.75:
            step_ms += random.randint(4, 14)
        page.wait_for_timeout(step_ms)

def human_move(page, x, y):
    if MOUSE["x"] is None:
        MOUSE["x"] = x + random.randint(-350, 350)
        MOUSE["y"] = y + random.randint(-250, 250)
        try:
            page.mouse.move(MOUSE["x"], MOUSE["y"])
        except Exception:
            pass
    p0 = (MOUSE["x"], MOUSE["y"])
    p2 = (x + random.uniform(-2, 2), y + random.uniform(-2, 2))
    dx, dy = p2[0] - p0[0], p2[1] - p0[1]
    dist = max(1.0, math.hypot(dx, dy))
    off = random.uniform(
        MOUSE_OFFSET_FACTOR - MOUSE_OFFSET_VARIANCE,
        MOUSE_OFFSET_FACTOR + MOUSE_OFFSET_VARIANCE) * dist
    if random.random() < 0.5:
        off = -off
    p1 = ((p0[0] + p2[0]) / 2 - dy / dist * off,
          (p0[1] + p2[1]) / 2 + dx / dist * off)
    _move_along(page, p0, p1, p2)
    if random.random() < MOUSE_OVERSHOOT_PROB:
        over = random.uniform(5, 16)
        ang = math.atan2(dy, dx)
        po = (p2[0] + math.cos(ang) * over + random.uniform(-3, 3),
              p2[1] + math.sin(ang) * over + random.uniform(-3, 3))
        _move_along(page, p2, p2, po)
        page.wait_for_timeout(random.randint(60, 140))
        _move_along(page, po, po, p2)
    MOUSE["x"], MOUSE["y"] = p2
    page.wait_for_timeout(
        random.randint(MOUSE_SETTLE_MIN_MS, MOUSE_SETTLE_MAX_MS))

def human_click(page, sel, aim="center"):
    loc = page.locator(sel).first
    try:
        loc.scroll_into_view_if_needed(timeout=SCROLL_INTO_VIEW_TIMEOUT)
    except Exception:
        pass
    box = None
    try:
        box = loc.bounding_box()
    except Exception:
        pass
    if not box:
        try:
            loc.click(timeout=CLICK_FALLBACK_TIMEOUT)
        except Exception:
            pass
        return
    if aim == "center":
        x = box["x"] + box["width"] * random.uniform(CLICK_AIM_MIN, CLICK_AIM_MAX)
        y = box["y"] + box["height"] * random.uniform(CLICK_AIM_MIN, CLICK_AIM_MAX)
    else:
        x = box["x"] + box["width"] / 2
        y = box["y"] + box["height"] / 2
    human_move(page, x, y)
    if random.random() < CLICK_AIM_HESITATE_PROB:
        page.wait_for_timeout(
            random.randint(CLICK_AIM_HESITATE_MIN, CLICK_AIM_HESITATE_MAX))
    else:
        page.wait_for_timeout(
            random.randint(CLICK_AIM_DELAY_MIN, CLICK_AIM_DELAY_MAX))
    try:
        page.mouse.down()
        page.wait_for_timeout(
            random.randint(CLICK_HOLD_MIN_MS, CLICK_HOLD_MAX_MS))
        page.mouse.up()
    except Exception:
        try:
            loc.click(timeout=CLICK_FALLBACK_TIMEOUT)
        except Exception:
            pass
        return
    MOUSE["x"], MOUSE["y"] = x, y
    page.wait_for_timeout(
        random.randint(CLICK_AFTER_MIN_MS, CLICK_AFTER_MAX_MS))

                pass

            if ch.isupper() or ch.isdigit() or not ch.isalnum():

    page.wait_for_timeout(random.randint(TYPE_AFTER_MIN, TYPE_AFTER_MAX))

def _native_keyboard_pick(page, sel, target_text):
    try:
        loc = page.locator(sel).first
        value = loc.evaluate(
            "el => el.selectedOptions[0]?.textContent?.trim() || ''")
        options = loc.evaluate(
            "el => Array.from(el.options).map(o => o.textContent.trim())")
        if target_text not in options:
            return False
        cur_idx = options.index(value) if value in options else 0
        target_idx = options.index(target_text)
        diff = target_idx - cur_idx
        if diff == 0:
            return True
        key = "ArrowDown" if diff > 0 else "ArrowUp"
        for _ in range(abs(diff)):
            page.keyboard.press(key)
            page.wait_for_timeout(
                random.randint(SELECT_ARROW_PAUSE_MIN, SELECT_ARROW_PAUSE_MAX))
        page.keyboard.press("Enter")
        page.wait_for_timeout(
            random.randint(SELECT_AFTER_PAUSE_MIN, SELECT_AFTER_PAUSE_MAX))
        dbg(f"keyboard-pick: '{target_text}' ({abs(diff)} шагов)")
        return True
    except Exception:
        return False

def pick_radix(page, container_sel, texts, label):
    for _attempt in (1, 2):
        if not radix_open(page, container_sel):
            close_leftover_popover(page)
            continue
        if not radix_click_option(page, texts, label):
            page.keyboard.press("Escape")
            page.wait_for_timeout(RADIX_RETRY_PAUSE)
            continue
        page.wait_for_timeout(random.randint(
            RADIX_CLICK_SETTLE_MIN, RADIX_CLICK_SETTLE_MAX))
        if radix_verify(page, container_sel, texts):
            return True
    close_leftover_popover(page)
    return False

def set_dropdown(page, sel, select_values, option_texts, label):
    loc = page.locator(sel).first
    try:
        tag = loc.evaluate("el => el.tagName.toLowerCase()")
    except Exception:
        log(f"⚠️  {label}: {sel} не найден", "WARN")
        return False
    if tag == "select":
        if select_native_human(page, sel, select_values, option_texts, label):
            return True
    else:
        if pick_radix(page, sel, option_texts, label):
            return True
    inner = f"{sel} select"
    try:
        if page.locator(inner).count() > 0:
            if select_option_hidden(page, inner, select_values,
                                    option_texts, label):
                return True
    except Exception:
        pass
    log(f"⚠️  {label}: не выбралось", "WARN")
    return False

def wait_submit_enabled(page, timeout=None):
    timeout = timeout or SUBMIT_WAIT_TIMEOUT_MS
    try:
        page.wait_for_function(
            """() => {
                const b = document.querySelector('#signup-button')
                       || document.querySelector("button[aria-label='Create account']")
                       || document.querySelector("button[type='submit']");
                return b && !b.disabled;
            }""",
            timeout=timeout)
        return True
    except Exception:
        return False

def click_submit(page):
    sels = [
        "#signup-button",
        "button[aria-label='Create account']",
        "button[type='submit']:not([disabled])",
        "button:has-text('Sign Up')",
        "button:has-text('Create account')",
    ]
    for sel in sels:
        try:
            loc = page.locator(sel).first
            if loc.is_visible() and loc.is_enabled():
                human_click(page, sel)
                return True
        except Exception:
            continue
    return False

def _session_body(browser, password, birth, gender, t0, exit_ip=None):

    page = browser.new_page()

        tz_ok, _bt, _pt = log_exit_ip(page, expected_ip=exit_ip)

            TZ_STATE["aborts"] += 1

                log("🛑 Прерываю сессию СРАЗУ — timezone не совпадает",
                    "ERROR")
                return False
            if not TZ_STATE["warned"]:
                TZ_STATE["warned"] = True

                    "запусти терминал от админа", "WARN")

            pass

              timeout=PAGE_LOAD_TIMEOUT)

        page.reload(wait_until="domcontentloaded")
        if not wait_for_form(page):
            log("❌ Форма не появилась", "ERROR")
            dump_html(page, "no-form")
            return False

        if has_pw_now else "v3 (multi-part)"
    log(f"  ✓ Форма готова ({variant}) | {time.time()-t0:.1f}с")
    shot(page, "00-landing")

    # ---- дата ----

                            [MONTH_ABBR[m]],
                            [MONTH_FULL[m], MONTH_ABBR[m]], "month")

                            [f"{d:02d}", str(d)], [str(d), f"{d:02d}"], "day")
        human_pause(page, PAUSE_BETWEEN_FIELDS_MIN, PAUSE_BETWEEN_FIELDS_MAX)
        ok_y = set_dropdown(page, "#YearDropdown",
                            [str(y)], [str(y)], "year")
        log(f"  {'✓' if ok_m and ok_d and ok_y else '⚠️ '} "
            f"Дата: m={ok_m} d={ok_d} y={ok_y} | {time.time()-t0:.1f}с")
        shot(page, "01-birthday")
    else:
        log("  → Даты нет, жму Next")
        advance_to_next_part(page)

    # ---- ник ----
    if not find_visible(page, USERNAME_SELECTORS):
        log("  → Ника нет, жму Next")
        if not advance_to_next_part(page):
            log("⚠️  Next не найден", "WARN")
        shot(page, "01b-after-next")
        if not wait_for_any_visible(
                page, USERNAME_SELECTORS, NEXT_WAIT_PART_TIMEOUT_S):
            log("❌ Поле ника не появилось", "ERROR")
            dump_html(page, "no-username")
            return False

    username = fill_username(page, birth)
    if username is None:
        log(f"❌ Ник не подобран за {USERNAME_MAX_TRIES} попыток", "ERROR")
        dump_html(page, "username-tries")
        return False
    log(f"  ✓ Ник: {username} | {time.time()-t0:.1f}с")

    # ---- пол: часть 1 ----
    human_pause(page, PAUSE_BETWEEN_FIELDS_MIN, PAUSE_BETWEEN_FIELDS_MAX)
    gender_done = select_gender(page, gender)
    if gender_done:
        log(f"  ✓ Пол (часть 1): {gender} | {time.time()-t0:.1f}с")

    # ---- пароль ----
    pw_sel = find_visible(page, PASSWORD_SELECTORS)
    if not pw_sel:
        log("  → Пароля нет (multi-part), жму Next")
        if not advance_to_next_part(page):
            log("⚠️  Next не найден", "WARN")
        shot(page, "01c-to-part2")
        pw_sel = wait_for_any_visible(
            page, PASSWORD_SELECTORS, NEXT_WAIT_PART_TIMEOUT_S)
        if not pw_sel:
            log("❌ Поле пароля не появилось", "ERROR")
            dump_html(page, "no-password")
            return False

    human_pause(page, PAUSE_BETWEEN_FIELDS_MIN, PAUSE_BETWEEN_FIELDS_MAX)
    try:
        human_type(page, pw_sel, password)
        log(f"  ✓ Password | {time.time()-t0:.1f}с")
    except Exception:
        log("⚠️  Password не заполнился", "WARN")
        dump_html(page, "password")

    if not gender_done:
        if select_gender(page, gender):
            log(f"  ✓ Пол (часть 2): {gender} | {time.time()-t0:.1f}с")
        else:
            log("  ✓ Пол пропущен", "WARN")

    shot(page, "02-filled")

    errs = read_validations(page)
    if "username" in errs:
        log(f"  ⚠️  Ошибка ника: {errs['username']}", "WARN")
        clear_field(page, USERNAME_SELECTORS[0])
        username = fill_username(page, birth)
        if username is None:
            return False

    wait_submit_enabled(page)

    human_pause(page, PAUSE_BEFORE_SUBMIT_MIN, PAUSE_BEFORE_SUBMIT_MAX)
    if not click_submit(page):
        try:
            page.locator(pw_sel).press("Enter")
        except Exception:
            log("⚠️  Сабмит вручную", "WARN")

    log("⏳ Жду капчу/результат...")

    success = False
    polls = CAPTCHA_WAIT_SECONDS * 1000 // CAPTCHA_POLL_INTERVAL_MS
    for i in range(polls):
        page.wait_for_timeout(CAPTCHA_POLL_INTERVAL_MS)
        if i % MODAL_POLL_EVERY == 0 and DISMISS_PASSKEY_MODAL:
            decline_passkey(page)
        if "createaccount" not in page.url.lower():
            success = True
            break
        try:
            if page.locator(REJECT_SELECTOR).count() > 0:
                log("🚫 Roblox отклонил регистрацию ('try again later')",
                    "ERROR")
                shot(page, "03-rejected")
                if DEBUG:
                    dump_html(page, "rejected")
                return False
        except Exception:
            pass

    if success:
        log(f"✅ Аккаунт зарегистрирован! URL: {page.url}")
        page.wait_for_timeout(random.randint(800, 1600))
        decline_passkey(page)
        shot(page, "03-success")

        token = get_roblosecurity(page) if SAVE_ROBLOSECURITY else None
        if token:
            log(f"🔑 .ROBLOSECURITY: {token[:40]}... ({len(token)} симв.)")
        else:
            log("⚠️  Кука .ROBLOSECURITY не найдена", "WARN")

        if SAVE_FULL_COOKIES:
            try:
                Path(COOKIES_DIR).mkdir(exist_ok=True)
                (Path(COOKIES_DIR) / f"{username}_cookies.json").write_text(
                    json.dumps(page.context.cookies(), indent=2),
                    encoding="utf-8")
            except Exception:
                pass

        save_account(username, password, birth, gender, token)
        return True

    log("❌ Не зарегистрировалось (таймаут ожидания).", "ERROR")
    errs = read_validations(page)
    if errs:
        log(f"   Ошибки: {errs}")
    shot(page, "03-timeout")
    return False


# Цепочка ослабления kwargs при ValueError browserforge.
# geoip — последний рубеж: датасет browserforge не имеет заголовков для
# части локалей (pl-PL/de-DE/...), которые geoip требует по стране IP.
RELAX_ORDER = ("screen", "window", "geoip")

def _run_session(base_kwargs, password, birth, gender, t0, exit_ip=None):
    attempt_kwargs = dict(base_kwargs)
    relax = list(RELAX_ORDER)
    same_retries = 0
    while True:
        body_started = False
        try:
            with Camoufox(**attempt_kwargs) as browser:
                body_started = True
                return _session_body(browser, password, birth, gender, t0,
                                     exit_ip=exit_ip)
        except TypeError as e:
            if body_started:
                raise
            mt = re.search(r"unexpected keyword argument '(\w+)'", str(e))
            if not mt or mt.group(1) not in attempt_kwargs:
                raise
            bad = mt.group(1)
            attempt_kwargs.pop(bad)
            dbg(f"Camoufox не знает '{bad}' — убираю")
            continue
        except ValueError as e:
            if body_started:
                raise
            if "No headers based on this input" not in str(e):
                raise
            same_retries += 1
            if same_retries <= 2:
                log(f"⚠️  browserforge: нет заголовков для локали страны IP "
                    f"— ретрай {same_retries}/2 (версия браузера случайна)",
                    "WARN")
                continue
            while relax and relax[0] not in attempt_kwargs:
                relax.pop(0)
            if not relax:
                raise
            bad = relax.pop(0)
            attempt_kwargs.pop(bad, None)
            if bad == "geoip":
                log("⚠️  browserforge: отключаю geoip — датасет не имеет "
                    "заголовков для локали этой страны", "WARN")
                log("   Timezone останется синхронизированным (явный kwarg "
                    "+ ОС), WebRTC убит. Locale станет нейтральной (en-US) "
                    "— слабый сигнал.", "WARN")
            else:
                log(f"⚠️  browserforge: убираю '{bad}' "
                    f"(конфликт с датасетом)", "WARN")
            continue
        except Exception as e:
            if body_started and _is_browser_crash(e):
                raise AccountCrash(f"{type(e).__name__}: {e}") from e
            raise


def create_account(index, total):
    t0 = time.time()
    password = random_password()
    birth = random_birth()
    gender = random_gender()
    m, d, y = birth["month"], birth["day"], birth["year"]
    TZ_STATE["aborts"] = 0
    TZ_STATE["warned"] = False

    for attempt in range(1, ACCOUNT_RETRIES + 1):
        if attempt > 1:
            pause = random.randint(RETRY_PAUSE_MIN_S, RETRY_PAUSE_MAX_S)
            log(f"⏸  Пауза перед ретраем {pause}с...")
            time.sleep(pause)

        if PROXY_ENABLED:
            try:
                proxy, session, exit_ip, geo = preflight_with_filter(
                    prev_ip=PREV_EXIT["ip"])
            except RotationDetected as rd:
                log("🚨 РОТАЦИЯ 'НА КАЖДЫЙ ЗАПРОС' ОБНАРУЖЕНА! "
                    f"({rd})", "ERROR")
                log("   → В ЛК поставь sticky-ротацию и перезапусти.",
                    "ERROR")
                return False
            if exit_ip:
                if PREV_EXIT["ip"] and exit_ip != PREV_EXIT["ip"]:
                    log(f"🆕 Новый IP для аккаунта: {exit_ip} "
                        f"(был {PREV_EXIT['ip']})")
                PREV_EXIT["ip"] = exit_ip
        else:
            proxy, session, exit_ip, geo = None, "-", None, None

        fp = make_fingerprint()
        sw, sh = fp["screen"]
        ww, wh = fp["window"]

        log("=" * 55)
        tag = f"попытка {attempt}/{ACCOUNT_RETRIES}" if attempt > 1 else "старт"
        log(f"АККАУНТ {index}/{total} | сессия {session} | {tag}")
        log(f"Пароль: {password} | Дата: {MONTH_FULL[m]} {d}, {y} | "
            f"Пол: {gender}")
        log(f"Железо: экран {sw}x{sh}, окно {ww}x{wh}, CPU {fp['cores']} ядер")
        if geo:
            city = geo.get("city") or "—"
            log(f"🌍 Exit: {city}, {geo.get('country')} "
                f"({geo.get('countryCode')}) | timezone: {geo.get('timezone')}")
        elif exit_ip:
            log(f"🌍 Exit-IP (preflight): {exit_ip} (гео недоступно)")
        log("=" * 55)

        # ═══ СИНХРОНИЗАЦИЯ СИСТЕМНОГО TIMEZONE ═══
        orig_tz = None
        expected_tz = None
        if geo and geo.get("timezone"):
            expected_tz = geo["timezone"]
        if SYNC_OS_TIMEZONE and proxy and expected_tz:
            orig_tz = get_os_timezone()
            if set_os_timezone(expected_tz):
                log(f"🖥  Системный timezone: {orig_tz} → {expected_tz}")
            else:
                orig_tz = None
                log("⚠️  TZ ОС не сменился — браузер будет с твоим "
                    "системным временем (утечка)", "ERROR")

        base_kwargs = dict(
            headless=HEADLESS, humanize=HUMANIZE, geoip=GEOIP,
            block_webrtc=BLOCK_WEBRTC, os=OS_LIST,
            window=(int(ww), int(wh)),
            hardware_concurrency=fp["cores"],
            firefox_user_prefs=FIREFOX_USER_PREFS,
            disable_coop=True,
            i_know_what_im_doing=True,
        )
        scr = make_screen(sw, sh)
        if scr is not None:
            base_kwargs["screen"] = scr

        if expected_tz:
            base_kwargs["timezone"] = expected_tz

        if MANUAL_LOCALE:
            base_kwargs["locale"] = [MANUAL_LOCALE]
        elif PASS_LOCALE_EXPLICITLY and geo and geo.get("countryCode"):
            loc = COUNTRY_LOCALE.get(geo["countryCode"])
            if loc:
                base_kwargs["locale"] = [loc]

        if proxy:
            base_kwargs["proxy"] = proxy

        ok = False
        try:
            ok = _run_session(base_kwargs, password, birth, gender, t0,
                              exit_ip=exit_ip)
        except AccountCrash as e:
            log(f"💥 Браузер крашнулся: {e}", "ERROR")
        except KeyboardInterrupt:
            if orig_tz:
                if restore_os_timezone(orig_tz):
                    log(f"🖥  Timezone ОС восстановлен: {orig_tz}")
                else:
                    log(f'🚨 НЕ ВЕРНУЛ TZ! Вручную: tzutil /s "{orig_tz}"',
                        "ERROR")
            raise
        finally:
            if orig_tz:
                if restore_os_timezone(orig_tz):
                    log(f"🖥  Timezone ОС восстановлен: {orig_tz}")
                else:
                    log(f'🚨 НЕ ВЕРНУЛ TZ! Вручную: tzutil /s "{orig_tz}"',
                        "ERROR")

        if ok:
            return True

        if attempt < ACCOUNT_RETRIES:
            log(f"↻ Попытка {attempt} не прошла — ретрай "
                f"({attempt + 1}/{ACCOUNT_RETRIES})")
            continue
        log(f"❌ Аккаунт {index}: не создан за {ACCOUNT_RETRIES} попыток",
            "ERROR")
        return False
    return False


def main():
    proxy_desc = (f"mobile pool {STATIC_SERVER}"
                  if PROXY_MODE == "static"
                  else f"ротация {PROXY_HOST}:{PROXY_PORT}")
    log(f"План: {ACCOUNTS_TO_CREATE} аккаунт(ов) | "
        f"прокси: {'ВКЛ — ' + proxy_desc if PROXY_ENABLED else 'выкл'} | "
        f"новый IP на аккаунт: {'ВКЛ' if NEW_IP_PER_ACCOUNT else 'выкл'} | "
        f"sync ОС-timezone: {'ВКЛ' if SYNC_OS_TIMEZONE else 'выкл'} | "
        f"geoip: {'ВКЛ' if GEOIP else 'выкл'} | "
        f"WebRTC: {'убит' if BLOCK_WEBRTC else 'жив'}")
    check_camoufox_version()
    if SYNC_OS_TIMEZONE and sys.platform == "win32":
        log(f"🖥  Текущий timezone Windows: {get_os_timezone()}")
    created = 0
    for i in range(1, ACCOUNTS_TO_CREATE + 1):
        try:
            if create_account(i, ACCOUNTS_TO_CREATE):
                created += 1
        except KeyboardInterrupt:
            log("⏹  Остановлено пользователем", "WARN")
            break
        except Exception as e:
            log(f"❌ Аккаунт {i} упал: {e.__class__.__name__}: {e}", "ERROR")
        if i < ACCOUNTS_TO_CREATE:
            pause = random.randint(ACCOUNT_PAUSE_MIN_S, ACCOUNT_PAUSE_MAX_S)
            log(f"⏸  Пауза {pause}с...")
            time.sleep(pause)
    log("=" * 55)
    log(f"Итог: создано {created}/{ACCOUNTS_TO_CREATE}")
    log("=" * 55)
    input("Enter для выхода...")


if __name__ == "__main__":
